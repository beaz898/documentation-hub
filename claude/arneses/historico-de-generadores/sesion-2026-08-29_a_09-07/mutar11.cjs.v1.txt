const fs = require('fs');
const { execSync } = require('child_process');

const P = 'lib/chunking.ts';
const original = fs.readFileSync(P, 'utf8');

const mutaciones = [
  { id: 'w1', que: 'el arranque vuelve a ser ciego (el estado de ayer)',
    de: '    start = arranqueEnFrontera(text, end - overlap, end);',
    a:  '    start = end - overlap;' },

  { id: 'w2', que: 'se busca la frontera MAS TARDIA en vez de la mas temprana (mata el solape)',
    de: '    const i = text.indexOf(sep, desde);',
    a:  '    const i = text.lastIndexOf(sep, hasta);' },

  { id: 'w3', que: 'se retira la preferencia por salto de linea (solo quedan parrafo y punto)',
    de: "  const candidatos: Array<[string, number]> = [['\\n\\n', 2], ['. ', 2], ['\\n', 1]];",
    a:  "  const candidatos: Array<[string, number]> = [['\\n\\n', 2], ['. ', 2]];" },

  { id: 'w4', que: 'se pierde la guarda de ventana: el arranque puede saltar mas alla del final',
    de: '    if (i !== -1 && i + salto < hasta) return i + salto;',
    a:  '    if (i !== -1) return i + salto;' },

  { id: 'w5', que: 'la guarda de desde<=0 desaparece (afecta al primer trozo)',
    de: '  if (desde <= 0 || desde >= hasta) return desde;',
    a:  '  if (desde >= hasta) return desde;' },
];

function correr() {
  try {
    return execSync('npx vitest run --reporter=verbose 2>&1',
      { encoding: 'utf8', maxBuffer: 60 * 1024 * 1024 });
  } catch (e) { return (e.stdout || '') + (e.stderr || ''); }
}

function resumen(out) {
  const m = out.match(/Tests\s+(?:(\d+) failed \| )?(\d+) passed/);
  const fallos = [...out.matchAll(/^\s*×.*$/gm)]
    .map(x => x[0].trim().replace(/\s+\d+ms$/, ''))
    .map(l => l.replace(/^× lib\/[a-z/-]+\.test\.ts > /, '').replace(/^× /, ''));
  return { linea: m ? m[0] : '??', fallos: [...new Set(fallos)] };
}

for (const mut of mutaciones) {
  const n = original.split(mut.de).length - 1;
  if (n !== 1) { console.log('\n### ' + mut.id + ' — NO APLICADA: patron x' + n); continue; }
  fs.writeFileSync(P, original.replace(mut.de, mut.a));
  const aplicada = !fs.readFileSync(P, 'utf8').includes(mut.de);
  const r = resumen(correr());
  console.log('\n### ' + mut.id + ' — ' + mut.que);
  console.log('  aplicada=' + aplicada + ' | ' + r.linea);
  if (r.fallos.length === 0) console.log('  SOBREVIVE');
  else r.fallos.slice(0, 6).forEach(f => console.log('  MUERE: ' + f));
  fs.writeFileSync(P, original);
}

fs.writeFileSync(P, original);
console.log('\n--- restaurado identico: ' + (fs.readFileSync(P, 'utf8') === original) + ' ---');
