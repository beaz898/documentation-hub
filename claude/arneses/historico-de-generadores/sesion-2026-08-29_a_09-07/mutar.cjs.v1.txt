for (const m of MUT) {
  const bruto = fs.readFileSync(m.f, 'utf8');
  const crlf = bruto.includes('\r\n');
  const lf = crlf ? bruto.split('\r\n').join('\n') : bruto;

  const veces = lf.split(m.de).length - 1;
  if (veces !== 1) { console.log('\n### ' + m.n + '\n  NO APLICADA - ocurrencias: ' + veces); continue; }

  const mutado = lf.replace(m.de, m.a);
  fs.writeFileSync(m.f, crlf ? mutado.split('\n').join('\r\n') : mutado);
  const cambiado = fs.readFileSync(m.f, 'utf8') !== bruto;

  let out = '';
  try { out = execSync('npx vitest run lib/analysis/recorrido-id-documento.test.ts 2>&1', { encoding: 'utf8' }); }
  catch (e) { out = (e.stdout || '') + (e.stderr || ''); }

  fs.writeFileSync(m.f, bruto);

  const fallos = [];
  const re = /[\u00d7]\s(.+)/g;
  let mm;
  while ((mm = re.exec(out)) !== null) fallos.push(mm[1].replace(/\s+\d+ms\s*$/, '').trim());
  const resumen = (out.match(/Tests\s+.*/) || ['(sin resumen)'])[0].trim();

  console.log('\n### ' + m.n);
  console.log('  mutacion aplicada: ' + cambiado);
  console.log('  ' + resumen);
  fallos.forEach(f => console.log('  x ' + f));
}
