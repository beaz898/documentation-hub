function correr() {
  try {
    const out = execSync('npx vitest run --reporter=verbose 2>&1', {
      encoding: 'utf8', maxBuffer: 40 * 1024 * 1024,
    });
    return out;
  } catch (e) {
    return (e.stdout || '') + (e.stderr || '');
  }
}

function resumen(out) {
  const m = out.match(/Tests\s+(?:(\d+) failed \| )?(\d+) passed/);
  const fallos = [...out.matchAll(/^\s*(?:×|FAIL).*$/gm)]
    .map(x => x[0].trim())
    .filter(l => l.startsWith('×'))
    .map(l => l.replace(/\s+\d+ms$/, ''));
  return { linea: m ? m[0] : '??', fallos: [...new Set(fallos)] };
}
