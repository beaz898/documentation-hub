// Mutantes de SQL_SEG_01: cada uno rompe una cosa del seguro; el arnés tiene que ponerse rojo con los tres tests
// cargados. Precomprobación de anclas, restauración en finally, huella y git status al final.
const fs = require('fs');
const crypto = require('crypto');
const { execSync } = require('child_process');
const F = 'E:/Doc-Hub/SQL_SEG_01_las_mediciones_en_csv.sql';
const original = fs.readFileSync(F, 'utf8');
const huella = s => crypto.createHash('sha256').update(s).digest('hex').slice(0, 16);
const h0 = huella(original);
const M = [
  ['sin el filtro de la organización', 'WHERE ar.org_id = p.org', 'WHERE true'],
  ['la descarga cortada a dos filas', 'ORDER BY ar.created_at, ar.id;', 'ORDER BY ar.created_at, ar.id LIMIT 2;'],
  ['al revés de fecha', 'ORDER BY ar.created_at, ar.id;', 'ORDER BY ar.created_at DESC, ar.id;'],
  ['el total por documento y no de todo', 'count(*) OVER () AS total_filas', 'count(*) OVER (PARTITION BY ar.document_name) AS total_filas'],
  ['los caracteres de un análisis recortado', 'length(ar.analysis::text) AS caracteres_del_analisis', 'length(left(ar.analysis::text, 1000)) AS caracteres_del_analisis'],
  ['la huella del análisis sobre los contadores', 'md5(ar.analysis::text) AS huella_del_analisis', 'md5(ar.pipeline_counters::text) AS huella_del_analisis'],
  ['el registro por llamada con otra clave', "(ar.analysis ? 'llamadasAlModelo')", "(ar.analysis ? 'llamadas')"],
  ['columnas escogidas en vez de la fila entera', '       ar.*\n', '       ar.id, ar.org_id, ar.user_id, ar.document_name, ar.analysis_type, ar.created_at, ar.analysis\n'],
];
const EOL = original.includes('\r\n') ? '\r\n' : '\n';
for (const m of M) { m[1] = m[1].split('\n').join(EOL); m[2] = m[2].split('\n').join(EOL); }
for (const [nombre, a] of M) {
  const k = original.split(a).length - 1;
  if (k !== 1) { console.error('ANCLA ' + k + ' veces: ' + nombre); process.exit(2); }
}
const res = [];
try {
  for (const [nombre, a, b] of M) {
    fs.writeFileSync(F, original.split(a).join(b));
    let out = '';
    try { out = execSync('npx vitest run --project arneses claude/arneses/seg', { cwd: 'E:/Doc-Hub', encoding: 'utf8', stdio: 'pipe' }); }
    catch (e) { out = (e.stdout || '') + (e.stderr || ''); }
    const t = /Tests\s+(.*)/.exec(out);
    const linea = t ? t[1].trim() : '(sin línea de Tests)';
    const rojo = /failed/.test(linea) && /\(3\)/.test(linea);
    res.push((rojo ? 'ROJO  ' : 'VIVO  ') + nombre + ' · ' + linea);
    console.log(res[res.length - 1]);
  }
} finally {
  fs.writeFileSync(F, original);
}
const h1 = huella(fs.readFileSync(F, 'utf8'));
console.log('huella antes ' + h0 + ' · después ' + h1 + (h0 === h1 ? ' · IGUAL' : ' · DISTINTA'));
console.log(execSync('git status --short', { cwd: 'E:/Doc-Hub', encoding: 'utf8' }));
console.log('rojos: ' + res.filter(r => r.startsWith('ROJO')).length + ' de ' + M.length);
